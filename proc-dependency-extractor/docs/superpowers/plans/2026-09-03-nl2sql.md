# NL2SQL 智能查数 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 中文业务问句 → GBase 8a 查询 SQL（生成 + JsqlParser 语法校验 + 失败重试），血缘图谱辅助选表召回与 JOIN 提示，前端新增"智能查数"页面，附 30 题题集跑批验收。

**架构：** 新增 nl2sql 模块：`SchemaLinkingService`（MySQL 元数据关键词召回 + Neo4j 字段反查宿主表）→ `JoinPathService`（候选表两两血缘 JOIN 提示）→ `LlmClient`（OpenAI 兼容公网 API 单次生成）→ `Nl2SqlServiceImpl` 编排 + `JsqlParser` 校验重试。规格见 `docs/superpowers/specs/2026-09-03-nl2sql-design.md`。

**技术栈：** Java 17 / Spring Boot 3.2.5（RestClient）、MyBatis-Plus、Neo4jClient、JsqlParser 4.9、JUnit5 + Mockito、Vue 3

实现 `SchemaLinkingServiceImpl.java`：

```java
package com.hb.extractor.service.impl;

import com.hb.extractor.config.Nl2SqlProperties;
import com.hb.extractor.dto.nl2sql.CandidateTable;
import com.hb.extractor.entity.TableColumnMeta;
import com.hb.extractor.mapper.TableColumnMetaMapper;
import com.hb.extractor.service.SchemaLinkingService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.data.neo4j.core.Neo4jClient;
import org.springframework.stereotype.Service;

import java.util.*;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
import java.util.stream.Collectors;

@Slf4j
@Service
public class SchemaLinkingServiceImpl implements SchemaLinkingService {

    private static final Pattern TOKEN_PATTERN = Pattern.compile("[A-Za-z_][A-Za-z0-9_]{1,}|\\d{2,}");

    @Autowired
    private TableColumnMetaMapper columnMetaMapper;

    @Autowired(required = false)
    private Neo4jClient neo4jClient;

    @Autowired
    private Nl2SqlProperties props;

    public SchemaLinkingServiceImpl() {}

    public SchemaLinkingServiceImpl(TableColumnMetaMapper mapper, Neo4jClient client, Nl2SqlProperties props) {
        this.columnMetaMapper = mapper;
        this.neo4jClient = client;
        this.props = props;
    }

    @Override
    public List<CandidateTable> link(String question) {
        if (question == null || question.isBlank()) return Collections.emptyList();
        List<String> grams = chineseBigrams(question);
        List<String> tokens = englishTokens(question);

        // 路 1：元数据匹配
        Map<String, CandidateTable> byTable = new LinkedHashMap<>();
        List<TableColumnMeta> hits = (grams.isEmpty() && tokens.isEmpty())
                ? Collections.emptyList()
                : columnMetaMapper.selectByCommentOrColumn(grams, tokens);
        for (TableColumnMeta c : hits) {
            if (c.getFullTableName() == null) continue;
            CandidateTable ct = byTable.computeIfAbsent(c.getFullTableName(), k -> {
                CandidateTable t = new CandidateTable();
                t.setFullTableName(c.getFullTableName());
                t.setSchemaName(c.getSchemaName());
                t.setFields(new ArrayList<>());
                t.setScore(0);
                return t;
            });
            CandidateTable.FieldBrief fb = new CandidateTable.FieldBrief();
            fb.setColumnName(c.getColumnName());
            fb.setColumnType(c.getColumnType());
            fb.setColumnComment(c.getColumnComment());
            ct.getFields().add(fb);
            // 权重：注释命中 3 > 字段名命中 2
            boolean commentHit = grams.stream().anyMatch(g -> c.getColumnComment() != null && c.getColumnComment().contains(g));
            boolean nameHit = tokens.stream().anyMatch(t -> c.getColumnName() != null && c.getColumnName().toLowerCase().contains(t));
            if (commentHit) { ct.setScore(ct.getScore() + 3); ct.setSource("COMMENT"); }
            else if (nameHit) { ct.setScore(ct.getScore() + 2); ct.setSource("COLUMN_NAME"); }
        }

        // 路 2：血缘反查（Neo4j 可用且路 1 命中不足时）
        if (neo4jClient != null && byTable.size() < props.getMaxCandidateTables()) {
            for (String token : tokens) {
                Collection<String> hosts = neo4jClient.query(
                        "MATCH (c:Column {name: $n})-[:BELONGS_TO]->(t:Table) " +
                        "RETURN DISTINCT t.name AS name LIMIT 5")
                        .bind(token.toUpperCase()).to("n")
                        .fetchAs(String.class).mappedBy((type, record) -> record.get("name").asString())
                        .all();
                for (String host : hosts) {
                    CandidateTable ct = byTable.computeIfAbsent(host, k -> {
                        CandidateTable t = new CandidateTable();
                        t.setFullTableName(host);
                        t.setFields(new ArrayList<>());
                        t.setScore(2);
                        t.setSource("LINEAGE");
                        return t;
                    });
                    if ("LINEAGE".equals(ct.getSource())) ct.setScore(ct.getScore() + 2);
                }
                if (byTable.size() >= props.getMaxCandidateTables()) break;
            }
        }

        // 裁剪：排序（得分降序）→ 截断表数 → 截断每表字段
        return byTable.values().stream()
                .sorted(Comparator.comparingDouble(CandidateTable::getScore).reversed())
                .limit(props.getMaxCandidateTables())
                .peek(ct -> {
                    if (ct.getFields().size() > props.getMaxFieldsPerTable()) {
                        ct.setFields(new ArrayList<>(ct.getFields().subList(0, props.getMaxFieldsPerTable())));
                    }
                })
                .collect(Collectors.toList());
    }

    /** 提取中文 2-gram（用于注释 LIKE 匹配） */
    static List<String> chineseBigrams(String question) {
        List<String> grams = new ArrayList<>();
        Matcher m = Pattern.compile("[\\u4e00-\\u9fa5]{2,}").matcher(question);
        // 滑动 2-gram，控制数量（取前 8 个，避免 SQL 过长）
        while (m.find()) {
            String seg = m.group();
            for (int i = 0; i + 2 <= seg.length() && grams.size() < 8; i++) {
                String g = seg.substring(i, i + 2);
                if (!grams.contains(g)) grams.add(g);
            }
        }
        return grams;
    }

    /** 提取英文/数字 token（字段名/疑似字段名），停用词过滤 */
    static List<String> englishTokens(String question) {
        Set<String> stop = Set.of("the", "a", "of", "in", "on", "and", "or", "to", "by", "is", "what", "how");
        List<String> tokens = new ArrayList<>();
        Matcher m = TOKEN_PATTERN.matcher(question);
        while (m.find()) {
            String t = m.group().toLowerCase();
            if (!stop.contains(t) && !tokens.contains(t)) tokens.add(t);
        }
        return tokens;
    }
}
```

```java
import org.springframework.data.neo4j.core.Neo4jClient;
import org.springframework.stereotype.Service;

import java.util.*;
import java.util.stream.Collectors;

@Slf4j
@Service
public class JoinPathServiceImpl implements JoinPathService {

    @Autowired(required = false)
    private Neo4jClient neo4jClient;

    public JoinPathServiceImpl() {}

    public JoinPathServiceImpl(Neo4jClient client) {
        this.neo4jClient = client;
    }

    @Override
    public List<JoinHint> findJoinPaths(List<String> tables) {
        if (tables == null || tables.size() < 2) return Collections.emptyList();
        List<JoinHint> hints = new ArrayList<>();
        for (int i = 0; i < tables.size(); i++) {
            for (int j = i + 1; j < tables.size(); j++) {
                String a = tables.get(i);
                String b = tables.get(j);
                // 1. 直接血缘边（同一 proc 中 A→B 或 B→A 的表级血缘）
                collectDirectEdge(a, b, hints);
                // 2. 1-2 跳共同上游
                collectCommonUpstream(a, b, hints);
            }
        }
        return hints;
    }

    private void collectDirectEdge(String a, String b, List<JoinHint> out) {
        String cypher =
            "MATCH (ta:Table {name: $a}), (tb:Table {name: $b}) " +
            "MATCH (ta)-[r1:READS|PRODUCES|CONSUMES*1..2]-(mid:Script)-[r2:READS|PRODUCES|CONSUMES*1..2]-(tb) " +
            "WHERE mid.file IS NOT NULL " +
            "RETURN mid.file AS proc LIMIT 3";
        Collection<Rec> recs = query(cypher, Map.of("a", a, "b", b));
        for (Rec r : recs) {
            JoinHint h = new JoinHint();
            h.setTableA(a); h.setTableB(b);
            h.setRelation("DIRECT_EDGE");
            h.setProcName(r.proc);
            out.add(h);
        }
    }

    private void collectCommonUpstream(String a, String b, List<JoinHint> out) {
        String cypher =
            "MATCH (ca:Column)-[:BELONGS_TO]->(ta:Table {name: $a}), " +
            "      (cb:Column)-[:BELONGS_TO]->(tb:Table {name: $b}) " +
            "WHERE ca.name = cb.name " +
            "RETURN DISTINCT ca.name AS fieldName LIMIT 10";
        Collection<Rec> recs = query(cypher, Map.of("a", a, "b", b));
        if (!recs.isEmpty()) {
            JoinHint h = new JoinHint();
            h.setTableA(a); h.setTableB(b);
            h.setRelation("COMMON_COLUMN");
            h.setJoinFields(recs.stream().map(r -> r.fieldName).collect(Collectors.toList()));
            out.add(h);
        }
    }

    private Collection<Rec> query(String cypher, Map<String, Object> params) {
        if (neo4jClient == null) return Collections.emptyList();
        try {
            Neo4jClient.RunnableSpecTightToDatabase spec = neo4jClient.query(cypher);
            for (Map.Entry<String, Object> e : params.entrySet()) {
                spec = spec.bind(e.getValue()).to(e.getKey());
            }
            return spec.fetchAs(Rec.class)
                    .mappedBy((type, record) -> {
                        Rec r = new Rec();
                        if (record.containsKey("proc")) r.proc = record.get("proc").asString(null);
                        if (record.containsKey("fieldName")) r.fieldName = record.get("fieldName").asString();
                        return r;
                    })
                    .all();
        } catch (Exception e) {
            log.warn("JOIN 路径查询失败 a={}, b={}: {}", params.get("a"), params.get("b"), e.getMessage());
            return Collections.emptyList();
        }
    }

    private static class Rec {
        String proc;
        String fieldName;
    }
}
```

package com.hb.extractor.service;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.hb.extractor.config.Nl2SqlProperties;
import com.hb.extractor.dto.nl2sql.CandidateTable;
import com.hb.extractor.dto.nl2sql.JoinHint;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.MediaType;
import org.springframework.stereotype.Service;
import org.springframework.web.client.RestClient;

import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.regex.Matcher;
import java.util.regex.Pattern;
import java.util.stream.Collectors;

/**
 * LLM 客户端：OpenAI 兼容 chat/completions + prompt 组装。
 * prompt 组装与响应解析为纯函数（可单测）；HTTP 调用经 #chat。
 */
@Slf4j
@Service
public class LlmClient {

    private static final Pattern JSON_BLOCK = Pattern.compile("```(?:json)?\\s*([\\s\\S]*?)```");
    private final ObjectMapper objectMapper = new ObjectMapper();

    @Autowired
    private Nl2SqlProperties props;

    public LlmClient() {}

    public LlmClient(Nl2SqlProperties props) {
        this.props = props;
    }

    /** 调 LLM 返回原始文本内容 */
    public String chat(String systemPrompt, String userPrompt) {
        RestClient restClient = RestClient.builder().baseUrl(props.getBaseUrl()).build();
        Map<String, Object> body = new HashMap<>();
        body.put("model", props.getModel());
        body.put("temperature", props.getTemperature());
        body.put("messages", List.of(
                Map.of("role", "system", "content", systemPrompt),
                Map.of("role", "user", "content", userPrompt)));
        body.put("response_format", Map.of("type", "json_object"));
        Map<?, ?> resp = restClient.post()
                .uri("/chat/completions")
                .header("Authorization", "Bearer " + props.getApiKey())
                .contentType(MediaType.APPLICATION_JSON)
                .body(body)
                .retrieve()
                .body(Map.class);
        if (resp == null) throw new RuntimeException("LLM 返回空响应");
        List<?> choices = (List<?>) resp.get("choices");
        if (choices == null || choices.isEmpty()) throw new RuntimeException("LLM 返回无 choices");
        Map<?, ?> message = (Map<?, ?>) ((Map<?, ?>) choices.get(0)).get("message");
        return String.valueOf(message.get("content"));
    }

    /** 系统 prompt：GBase 8a 方言约束（沉淀自项目 SqlPlaceholderReplacer 经验） */
    public String buildSystemPrompt() {
        return """
                你是 GBase 8a 数据库的 SQL 专家。根据用户问题和候选表结构生成查询 SQL。

                方言约束（必须遵守）：
                1. 目标方言是 GBase 8a（兼容大部分 MySQL 语法），不要使用 GBase 不支持的语法。
                2. 日期字面量写 DATE'YYYY-MM-DD' 形式（无空格），不要用 STR_TO_DATE。
                3. 集合操作仅支持 UNION [ALL] / EXCEPT [ALL] / INTERSECT [ALL]。
                4. 分页用 LIMIT n OFFSET m。
                5. 只允许使用候选表清单中列出的表和字段，禁止猜测不存在的字段。
                6. schema 层级提示：cdm/fdm/odm 为明细层，dwa/pid 为汇总层（优先选汇总层减少数据量，除非问题明确要明细）。

                输出严格的 JSON 对象，格式：
                {"sql": "生成的SQL", "tables_used": ["用到的表"], "join_notes": ["JOIN说明"], "assumptions": ["你做的假设"]}
                """;
    }

    /** 用户 prompt：候选 schema + JOIN 提示 + 问句 */
    public String buildUserPrompt(List<CandidateTable> candidates, List<JoinHint> joinHints, String question) {
        StringBuilder sb = new StringBuilder();
        sb.append("候选表结构（JSON）：\n");
        String schemaJson = candidates.stream().map(t -> {
            Map<String, Object> m = new HashMap<>();
            m.put("table", t.getFullTableName());
            m.put("schema", t.getSchemaName());
            m.put("fields", t.getFields().stream().map(f -> {
                Map<String, String> fm = new HashMap<>();
                fm.put("name", f.getColumnName());
                fm.put("type", f.getColumnType());
                fm.put("comment", f.getColumnComment());
                return fm;
            }).collect(Collectors.toList()));
            return m;
        }).collect(Collectors.toList());
        sb.append(toJson(schemaJson)).append("\n\n");
        if (joinHints != null && !joinHints.isEmpty()) {
            sb.append("表间血缘关联提示（来自血缘分析，可信）：\n");
            for (JoinHint h : joinHints) {
                if ("COMMON_COLUMN".equals(h.getRelation())) {
                    sb.append(String.format("- %s 与 %s 存在共同字段 %s，可作为 JOIN 条件%n",
                            h.getTableA(), h.getTableB(), h.getJoinFields()));
                } else {
                    sb.append(String.format("- %s 与 %s 曾在同一存储过程 %s 中共同出现（血缘关联）%n",
                            h.getTableA(), h.getTableB(), h.getProcName()));
                }
            }
            sb.append("\n");
        }
        sb.append("用户问题：").append(question);
        return sb.toString();
    }

    /** 从 LLM 回复中提取 JSON（兼容 ```json 代码块包裹） */
    public JsonNode extractJson(String content) {
        String json = content;
        Matcher m = JSON_BLOCK.matcher(content);
        if (m.find()) json = m.group(1).trim();
        json = json.trim();
        if (!json.startsWith("{")) {
            throw new RuntimeException("LLM 返回内容不是 JSON：" + content.substring(0, Math.min(100, content.length())));
        }
        try {
            return objectMapper.readTree(json);
        } catch (Exception e) {
            throw new RuntimeException("LLM 返回 JSON 解析失败：" + e.getMessage());
        }
    }

    private String toJson(Object o) {
        try {
            return objectMapper.writeValueAsString(o);
        } catch (Exception e) {
            return "[]";
        }
    }
}

```java
package com.hb.extractor.service.impl;

import com.fasterxml.jackson.databind.JsonNode;
import com.hb.extractor.config.Nl2SqlProperties;
import com.hb.extractor.dto.nl2sql.CandidateTable;
import com.hb.extractor.dto.nl2sql.JoinHint;
import com.hb.extractor.dto.nl2sql.Nl2SqlResult;
import com.hb.extractor.service.JoinPathService;
import com.hb.extractor.service.LlmClient;
import com.hb.extractor.service.Nl2SqlService;
import com.hb.extractor.service.SchemaLinkingService;
import lombok.extern.slf4j.Slf4j;
import net.sf.jsqlparser.parser.CCJSqlParserUtil;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.ArrayList;
import java.util.List;
import java.util.stream.Collectors;
import java.util.stream.StreamSupport;

@Slf4j
@Service
public class Nl2SqlServiceImpl implements Nl2SqlService {

    @Autowired
    private SchemaLinkingService schemaLinkingService;
    @Autowired
    private JoinPathService joinPathService;
    @Autowired
    private LlmClient llmClient;
    @Autowired
    private Nl2SqlProperties props;

    public Nl2SqlServiceImpl() {}

    public Nl2SqlServiceImpl(SchemaLinkingService linking, JoinPathService joinPath,
                             LlmClient llm, Nl2SqlProperties props) {
        this.schemaLinkingService = linking;
        this.joinPathService = joinPath;
        this.llmClient = llm;
        this.props = props;
    }

    @Override
    public Nl2SqlResult generate(String question) {
        Nl2SqlResult result = new Nl2SqlResult();
        result.setQuestion(question);

        // ① 召回
        List<CandidateTable> candidates = schemaLinkingService.link(question);
        result.setCandidates(candidates);
        if (candidates.isEmpty()) {
            result.setValidationStatus("ERROR");
            result.setErrorMessage("未匹配到任何表：请换更贴近字段注释/表名的措辞重试");
            return result;
        }

        // ② JOIN 提示
        List<JoinHint> hints = joinPathService.findJoinPaths(
                candidates.stream().map(CandidateTable::getFullTableName).collect(Collectors.toList()));
        result.setLineageEvidence(hints);

        // ③④ 生成 + 校验重试
        String systemPrompt = llmClient.buildSystemPrompt();
        String userPrompt = llmClient.buildUserPrompt(candidates, hints, question);
        String lastError = null;
        String lastSql = null;
        for (int attempt = 0; attempt <= props.getMaxRetries(); attempt++) {
            try {
                String prompt = (lastError == null) ? userPrompt
                        : userPrompt + "\n\n上一次生成的 SQL 语法校验失败：" + lastSql + "\n报错：" + lastError + "\n请修正。";
                String content = llmClient.chat(systemPrompt, prompt);
                JsonNode node = llmClient.extractJson(content);
                String sql = node.path("sql").asText();
                String err = validate(sql);
                if (err == null) {
                    result.setSql(sql);
                    result.setTablesUsed(toStringList(node.path("tables_used")));
                    result.setJoinNotes(toStringList(node.path("join_notes")));
                    result.setAssumptions(toStringList(node.path("assumptions")));
                    result.setValidationStatus("VALIDATED");
                    return result;
                }
                lastError = err;
                lastSql = sql;
            } catch (Exception e) {
                log.warn("NL2SQL 第 {} 次生成失败: {}", attempt + 1, e.getMessage());
                lastError = e.getMessage();
            }
        }
        result.setValidationStatus("FAILED");
        result.setErrorMessage(lastError != null ? lastError : "生成失败");
        return result;
    }

    /** JsqlParser 语法校验；通过返回 null，失败返回错误信息 */
    String validate(String sql) {
        if (sql == null || sql.isBlank()) return "SQL 为空";
        try {
            CCJSqlParserUtil.parse(sql);
            return null;
        } catch (Exception e) {
            return e.getMessage();
        }
    }

    private List<String> toStringList(JsonNode node) {
        if (node == null || !node.isArray()) return new ArrayList<>();
        return StreamSupport.stream(node.spliterator(), false)
                .map(JsonNode::asText).collect(Collectors.toList());
    }
}
```

```java
package com.hb.extractor.controller;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.hb.extractor.config.Nl2SqlProperties;
import com.hb.extractor.dto.Result;
import com.hb.extractor.dto.nl2sql.Nl2SqlResult;
import com.hb.extractor.service.Nl2SqlService;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;

import java.io.File;
import java.util.*;

/**
 * NL2SQL 智能查数 Controller
 */
@Slf4j
@RestController
@RequestMapping("/api/nl2sql")
public class Nl2SqlController {

    @Autowired
    private Nl2SqlService nl2SqlService;
    @Autowired
    private Nl2SqlProperties props;

    private final ObjectMapper objectMapper = new ObjectMapper();

    /**
     * 生成 SQL
     * POST /api/nl2sql/generate  body: {"question": "..."}
     */
    @PostMapping("/generate")
    public Result<Nl2SqlResult> generate(@RequestBody Map<String, String> body) {
        String question = body.get("question");
        if (question == null || question.isBlank()) {
            return Result.error("question 不能为空");
        }
        return Result.success(nl2SqlService.generate(question));
    }

    /**
     * 题集跑批评估
     * POST /api/nl2sql/eval
     * 输出：总题数 / 选表正确率 / 语法通过率 / 整体正确率 / 明细
     */
    @PostMapping("/eval")
    public Result<Map<String, Object>> eval() throws Exception {
        File file = new File(props.getEvalFile());
        if (!file.exists()) {
            return Result.error("题集文件不存在: " + props.getEvalFile());
        }
        JsonNode questions = objectMapper.readTree(file).get("questions");
        int total = 0, tableCorrect = 0, syntaxPass = 0, overall = 0;
        List<Map<String, Object>> details = new ArrayList<>();
        for (JsonNode q : questions) {
            total++;
            String question = q.get("question").asText();
            Set<String> expected = new HashSet<>();
            q.get("expected_tables").forEach(t -> expected.add(t.asText().toLowerCase()));
            Nl2SqlResult r = nl2SqlService.generate(question);

            Set<String> used = new HashSet<>();
            if (r.getTablesUsed() != null) r.getTablesUsed().forEach(t -> used.add(t.toLowerCase()));
            boolean tablesOk = !used.isEmpty() && expected.containsAll(used) && used.containsAll(expected);
            boolean syntaxOk = "VALIDATED".equals(r.getValidationStatus());
            if (tablesOk) tableCorrect++;
            if (syntaxOk) syntaxPass++;
            if (tablesOk && syntaxOk) overall++;

            Map<String, Object> d = new LinkedHashMap<>();
            d.put("question", question);
            d.put("expectedTables", expected);
            d.put("usedTables", used);
            d.put("tablesCorrect", tablesOk);
            d.put("syntaxValid", syntaxOk);
            d.put("status", r.getValidationStatus());
            d.put("sql", r.getSql());
            d.put("error", r.getErrorMessage());
            details.add(d);
        }
        Map<String, Object> report = new LinkedHashMap<>();
        report.put("total", total);
        report.put("tableAccuracy", total == 0 ? 0 : (double) tableCorrect / total);
        report.put("syntaxPassRate", total == 0 ? 0 : (double) syntaxPass / total);
        report.put("overallAccuracy", total == 0 ? 0 : (double) overall / total);
        report.put("details", details);
        return Result.success(report);
    }
}
```

```vue
<script setup>
import { ref, computed } from 'vue'
import { generate } from '../api/nl2sql'
import { ElMessage } from 'element-plus'

const question = ref('')
const loading = ref(false)
const result = ref(null)
const error = ref('')

const statusType = computed(() => {
  if (!result.value) return 'info'
  if (result.value.validationStatus === 'VALIDATED') return 'success'
  if (result.value.validationStatus === 'FAILED') return 'warning'
  return 'danger'
})
const statusText = computed(() => {
  const map = { VALIDATED: '语法校验通过', FAILED: '语法校验未通过（已重试）', ERROR: '生成失败' }
  return map[result.value?.validationStatus] || '未知'
})

async function doGenerate() {
  if (!question.value.trim()) {
    ElMessage.warning('请输入问题')
    return
  }
  loading.value = true
  error.value = ''
  result.value = null
  try {
    const res = await generate({ question: question.value.trim() })
    result.value = res.data
  } catch (e) {
    error.value = '调用失败：' + (e.message || e)
  } finally {
    loading.value = false
  }
}
</script>
```
